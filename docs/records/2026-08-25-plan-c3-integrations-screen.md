**Goal:** Ship Epic C delivery step 3 (spec `docs/superpowers/specs/2026-08-22-epic-c-social-accounts-design.md`, §6 item 3, part of #642 / #630): the Settings → Integrations page — connected-account list (handle, provider, status pill green/orange/grey, last success, "Visible in: all projects" or project names), the Connect Bluesky drawer (identifier + app password + help link, provider refusals shown as-is), Attach-to-projects checklist (none checked = all projects), Reconnect (same drawer, handle prefilled), Disconnect with a consequence-naming confirmation, and the persistent workspace `NeedsReconnect` banner whose Reconnect button renders only for holders of `tenant.socialaccounts.manage`. i18n in both locales, component tests for drawer + banner, one e2e tagged per #1168, permissions-driven visibility, query invalidation after every action, every failure carrying its cause per DESIGN.md error states.

## Global Constraints

Copied from spec §7 ("Hard repo constraints that apply") as they touch this lane, plus the standing front rules:

- Vertical slice stays under `apps/api/Modules/SocialAccounts` on the C2 branch — **this plan writes zero backend code**; any contract gap is recorded as an assumption, never invented here.
- The app secret (Bluesky app password) is never returned by any API, never logged, never rendered, never stored in component state beyond the form field lifetime, never in an error message.
- Errors shown as-is with their cause: `credentials_refused` / account-not-found / `provider_unreachable` map to plain-words copy that names what went wrong (DESIGN.md error states; owner rule "transparent failure causes"). No generic "something went wrong".
- Only `401` triggers centralized logout; `403` never logs out — a view-less user gets the Integrations nav hidden, and a denied action surfaces the problem detail message without logout.
- URL/query parameter names snake_case; JSON fields camelCase; no collapsed lowercase wire values (`needs_reconnect`, never `needsreconnect`).
- i18n keys added to **both** locales: `apps/front/src/i18n/locales/en/settings.json` AND `apps/front/src/i18n/locales/fr/settings.json`. The key-parity suite `apps/front/src/lib/i18n-key-coverage.test.ts` fails on any missing fr/en counterpart.
- Design-token discipline is machine-checked (`pnpm --filter front check:design-system` runs inside tests): status pills use the existing `Badge` variants; no raw hex; every z-index utility through the `--publy-z-*` scale; dark mode via tokens only.
- Lint hard rules: no `Array.reduce`, no IIFEs, arrow-function components only (`publy/arrow-function-components` at error), no direct dayjs imports in components, specific-lodash imports, no `console.*` in source, no manual translation of `response-message` keys at call sites (`getFailureMessage(toApiFailure(error), ...)` only).
- React Doctor HARD gate: `just react-doctor` clean in every file this plan touches before each push.
- Forms go through React Hook Form + Zod via the front field wrappers; the drawer form follows the `drawer-form.test.tsx` patterns.
- Loading/empty/error states use `state-surface.tsx` / `skeleton.tsx` — never ad-hoc conditionals.
- The e2e fakes Bluesky (never the real network) and is tagged `@tenant-workspace @642` per the #1168 vocabulary; it lives with the other tenant e2e specs and runs only in the e2e suite, not `just ci`.
- Secrets never in output; never spawn sub-agents; commit early, push after every commit; never touch develop.

// apps/front/src/lib/permissions/use-has-tenant-permission.ts
import { useMemo } from 'react';

import { useCurrentUserQuery } from '~/lib/query/auth';

const SOCIAL_VIEW_PERMISSION = 'tenant.socialaccounts.view';

/** Permission gate for UI affordances. Defaults to `false` while the session
 * loads — actions appear late, never act without the right. */
export const useHasTenantPermission = (key: string): boolean => {
	const { data } = useCurrentUserQuery();

	return useMemo(() => {
		if (!data || data.tenantPermissionKeys.length === 0) {
			return false;
		}

		return data.tenantPermissionKeys.includes(key);
	}, [data, key]);
};

/** Convenience wrapper for the social slice. */
export const useCanManageSocialAccounts = (): boolean =>
	useHasTenantPermission('tenant.socialaccounts.manage');

export const useCanViewIntegrations = (): boolean =>
	useHasTenantPermission(SOCIAL_VIEW_PERMISSION);

import type { QueryClient } from '@tanstack/react-query';
import { useMutation, useQuery, useQueryClient } from '@tanstack/react-query';
import { getClientManager } from '~/lib/api-client/client-manager';
import {

	scopedKey,

import type { ApiClient } from '@org/client-ts/apiClient';

/** Wire statuses come from C2's `SocialAccountWire.FormatStatus`
 * (branch origin/lane/wt-641): `active | needs_reconnect | revoked`.
 * Tones follow Epic C §3 exactly — green active, orange needs reconnect,
 * GREY revoked. Deliberately NOT `statusPillTone()` from
 * components/ui/status-tone.ts: it maps `revoked` → danger(red), which the
 * spec overrides for this screen (revocation is a state, not an error). */
export type SocialAccountStatusWire = 'active' | 'needs_reconnect' | 'revoked';

const TONES: Record<SocialAccountStatusWire, 'success' | 'warning' | 'neutral'> = {
	active: 'success',
	needs_reconnect: 'warning',
	revoked: 'neutral',
};

const LABEL_KEYS: Record<SocialAccountStatusWire, string> = {
	active: 'status-active',
	needs_reconnect: 'status-needs-reconnect',
	revoked: 'status-revoked',
};

export type SocialAccountRow = {
	id: string;
	displayHandle: string;
	statusWire: SocialAccountStatusWire;
	tone: 'success' | 'warning' | 'neutral';
	statusLabelKey: string;
	lastSuccessAt: Date | null;
	projectIds: string[];
};

export type SocialAccountsResponse = {
	value: Array<{
		id: string;
		displayHandle: string;
		provider: string;
		status: string;
		lastSuccessAt: string | null;
		projectIds: string[];
	}>;
};

const toWireStatus = (status: string | null | undefined): SocialAccountStatusWire =>
	status === 'needs_reconnect' || status === 'revoked' ? status : 'active';

const toDate = (value: string | null | undefined): Date | null => {
	if (!value) return null;
	const parsed = new Date(value);
	return Number.isNaN(parsed.valueOf()) ? null : parsed;
};

export const SOCIAL_ACCOUNTS_QUERY_KEY = ['tenant-social-accounts'] as const;

export const toSocialAccountRows = (
	response: SocialAccountsResponse | null | undefined,
): SocialAccountRow[] => {
	if (!response?.value) return [];

	return response.value.map((item) => {
		const statusWire = toWireStatus(item.status);
		return {
			id: item.id,
			displayHandle: item.displayHandle ?? '',
			statusWire,
			tone: TONES[statusWire],
			statusLabelKey: LABEL_KEYS[statusWire],
			lastSuccessAt: toDate(item.lastSuccessAt),
			projectIds: Array.isArray(item.projectIds) ? item.projectIds : [],
		};
	});
};

export const socialAccountsQueryOptions = buildTenantQueryOptions<
	ApiClient,
	SocialAccountsResponse,
	Record<string, never>
>(
	{
		queryKeyFn: () => [...SOCIAL_ACCOUNTS_QUERY_KEY],

			const result = await client.socialAccounts.get(); // [A3] name pinned post-regeneration
