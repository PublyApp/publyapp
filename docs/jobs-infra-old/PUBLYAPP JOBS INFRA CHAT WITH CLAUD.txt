public class PostSchedulingService
{
    private readonly ISchedulerFactory _schedulerFactory;

    public PostSchedulingService(ISchedulerFactory schedulerFactory)
    {
        _schedulerFactory = schedulerFactory;
    }

    public async Task SchedulePostAsync(Guid postId, Guid tenantId, DateTimeOffset publishAt)
    {
        var scheduler = await _schedulerFactory.GetScheduler();

        var jobKey = new JobKey($"post-{postId}", "scheduled-posts");
        var triggerKey = new TriggerKey($"post-{postId}-trigger", "scheduled-posts");

        var job = JobBuilder.Create<PublishPostJob>()
            .WithIdentity(jobKey)
            .UsingJobData("PostId", postId.ToString())
            .UsingJobData("TenantId", tenantId.ToString())
            .Build();

        var trigger = TriggerBuilder.Create()
            .WithIdentity(triggerKey)
            .StartAt(publishAt)
            .Build();

        await scheduler.ScheduleJob(job, trigger);
    }

    public async Task ReschedulePostAsync(Guid postId, DateTimeOffset newPublishAt)
    {
        var scheduler = await _schedulerFactory.GetScheduler();
        var triggerKey = new TriggerKey($"post-{postId}-trigger", "scheduled-posts");

        var newTrigger = TriggerBuilder.Create()
            .WithIdentity(triggerKey)
            .StartAt(newPublishAt)
            .Build();

        await scheduler.RescheduleJob(triggerKey, newTrigger);
    }

    public async Task CancelScheduledPostAsync(Guid postId)
    {
        var scheduler = await _schedulerFactory.GetScheduler();
        var jobKey = new JobKey($"post-{postId}", "scheduled-posts");
        
        await scheduler.DeleteJob(jobKey);
    }

    public async Task<DateTimeOffset?> GetScheduledTimeAsync(Guid postId)
    {
        var scheduler = await _schedulerFactory.GetScheduler();
        var triggerKey = new TriggerKey($"post-{postId}-trigger", "scheduled-posts");
        
        var trigger = await scheduler.GetTrigger(triggerKey);
        return trigger?.StartTimeUtc;
    }

Error Handling & Retries
Quartz doesn't have built-in automatic retries like Hangfire, but you can implement them:

csharp
public class PublishPostJob : IJob
{
    private const int MaxRetries = 3;

    public async Task Execute(IJobExecutionContext context)
    {
        var retryCount = context.JobDetail.JobDataMap.GetInt("RetryCount");
        
        try
        {
            await DoPublishAsync(context);
        }
        catch (Exception ex) when (retryCount < MaxRetries)
        {
            // Reschedule with incremented retry count
            var scheduler = context.Scheduler;
            var jobData = context.JobDetail.JobDataMap;
            jobData.Put("RetryCount", retryCount + 1);

            var delay = TimeSpan.FromMinutes(Math.Pow(2, retryCount)); // Exponential backoff
            
            var newTrigger = TriggerBuilder.Create()
                .StartAt(DateTimeOffset.UtcNow.Add(delay))
                .Build();

            await scheduler.RescheduleJob(context.Trigger.Key, newTrigger);
            
            throw new JobExecutionException(ex, refireImmediately: false);
        }
    }
}
Or use a Job Listener for centralized retry logic:

csharp
public class RetryJobListener : IJobListener
{
    public string Name => "RetryJobListener";

    public Task JobToBeExecuted(IJobExecutionContext context, 
        CancellationToken ct) => Task.CompletedTask;

    public Task JobExecutionVetoed(IJobExecutionContext context, 
        CancellationToken ct) => Task.CompletedTask;

    public async Task JobWasExecuted(IJobExecutionContext context, 
        JobExecutionException? jobException, CancellationToken ct)
    {
        if (jobException == null) return;

        var retryCount = context.JobDetail.JobDataMap.GetInt("RetryCount");
        if (retryCount >= 3) return; // Max retries reached

        // Schedule retry with backoff
        var delay = TimeSpan.FromSeconds(30 * Math.Pow(2, retryCount));
        context.JobDetail.JobDataMap.Put("RetryCount", retryCount + 1);

        var trigger = TriggerBuilder.Create()
            .ForJob(context.JobDetail)
            .StartAt(DateTimeOffset.UtcNow.Add(delay))
            .Build();

        await context.Scheduler.ScheduleJob(trigger);
    }
}

public class RedisJobScheduler : BackgroundService
{
    private readonly IConnectionMultiplexer _redis;
    private readonly IServiceProvider _services;
    private const string ScheduledJobsKey = "publyapp:scheduled-jobs";

    protected override async Task ExecuteAsync(CancellationToken stoppingToken)
    {
        var db = _redis.GetDatabase();
        
        while (!stoppingToken.IsCancellationRequested)
        {
            // Get jobs due now (score = Unix timestamp)
            var now = DateTimeOffset.UtcNow.ToUnixTimeSeconds();
            var dueJobs = await db.SortedSetRangeByScoreAsync(
                ScheduledJobsKey, 
                0, 
                now, 
                take: 10);

            foreach (var job in dueJobs)
            {
                var jobData = JsonSerializer.Deserialize<ScheduledJob>(job!);
                
                // Process job
                using var scope = _services.CreateScope();
                var handler = scope.ServiceProvider
                    .GetRequiredService<IJobHandler>();
                await handler.HandleAsync(jobData);
                
                // Remove from sorted set
                await db.SortedSetRemoveAsync(ScheduledJobsKey, job);
            }
            
            await Task.Delay(1000, stoppingToken); // Poll every second
        }
    }

    public async Task ScheduleAsync(string jobType, object data, DateTimeOffset runAt)
    {
        var db = _redis.GetDatabase();
        var job = new ScheduledJob(jobType, JsonSerializer.Serialize(data));
        
        await db.SortedSetAddAsync(
            ScheduledJobsKey,
            JsonSerializer.Serialize(job),
            runAt.ToUnixTimeSeconds());
    }
}

public static class PostHandlers
{
    public static async Task Handle(
        PublishPost command,
        AppDbContext db,
        ILinkedInClient linkedIn,
        ILogger<PublishPost> logger,
        CancellationToken ct)
    {
        logger.LogInformation("Publishing post {PostId} for tenant {TenantId}",
            command.PostId, command.TenantId);

        var post = await db.ScheduledPosts
            .Include(p => p.SocialAccount)
            .FirstOrDefaultAsync(p => p.Id == command.PostId && p.TenantId == command.TenantId, ct);

        if (post is null)
        {
            logger.LogWarning("Post {PostId} not found or already published", command.PostId);
            return;
        }

        if (post.Status != PostStatus.Scheduled)
        {
            logger.LogWarning("Post {PostId} is not in Scheduled status", command.PostId);
            return;
        }

        try
        {
            var result = await linkedIn.PublishAsync(
                post.SocialAccount.AccessToken,
                post.Content,
                post.ImageUrl,
                ct);

            post.Status = PostStatus.Published;
            post.PublishedAt = DateTime.UtcNow;
            post.PlatformPostId = result.PostId;

            await db.SaveChangesAsync(ct);
            
            logger.LogInformation("Successfully published post {PostId}", command.PostId);
        }
        catch (Exception ex)
        {
            post.Status = PostStatus.Failed;
            post.FailureReason = ex.Message;
            post.RetryCount++;
            
            await db.SaveChangesAsync(ct);
            
            throw; // Let Wolverine handle retry
        }
    }
}

// Features/Posts/PostSchedulingService.cs
public class PostSchedulingService
{
    private readonly IMessageBus _bus;
    private readonly AppDbContext _db;

    public PostSchedulingService(IMessageBus bus, AppDbContext db)
    {
        _bus = bus;
        _db = db;
    }

    public async Task SchedulePostAsync(
        Guid postId, 
        Guid tenantId, 
        DateTimeOffset publishAt,
        CancellationToken ct = default)
    {
        // Save to DB first
        var post = await _db.ScheduledPosts.FindAsync([postId], ct);
        if (post is null) throw new NotFoundException("Post not found");

        post.ScheduledFor = publishAt.UtcDateTime;
        post.Status = PostStatus.Scheduled;
        await _db.SaveChangesAsync(ct);

        // Schedule the message
        await _bus.ScheduleAsync(
            new PublishPost(postId, tenantId),
            publishAt);
    }

    public async Task ReschedulePostAsync(
        Guid postId,
        Guid tenantId,
        DateTimeOffset newPublishAt,
        CancellationToken ct = default)
    {
        // Update DB
        var post = await _db.ScheduledPosts.FindAsync([postId], ct);
        if (post is null) throw new NotFoundException("Post not found");

        post.ScheduledFor = newPublishAt.UtcDateTime;
        await _db.SaveChangesAsync(ct);

        // Cancel old + schedule new
        // Note: Wolverine doesn't have built-in cancel by correlation ID
        // We handle this by checking post status in the handler
        await _bus.ScheduleAsync(
            new PublishPost(postId, tenantId),
            newPublishAt);
    }

    public async Task CancelScheduledPostAsync(Guid postId, CancellationToken ct = default)
    {
        var post = await _db.ScheduledPosts.FindAsync([postId], ct);
        if (post is null) return;

        post.Status = PostStatus.Draft;
        post.ScheduledFor = null;
        await _db.SaveChangesAsync(ct);
        
        // The handler checks status, so even if message fires, it won't publish
    }
}
Recurring Jobs (System Tasks)
Wolverine doesn't have built-in cron scheduling like Hangfire, but you can implement it cleanly with a hosted service:

csharp
// Infrastructure/Scheduling/RecurringJobsHostedService.cs
public class RecurringJobsHostedService : BackgroundService
{
    private readonly IServiceProvider _services;
    private readonly ILogger<RecurringJobsHostedService> _logger;
    private readonly List<RecurringJobDefinition> _jobs;

    public RecurringJobsHostedService(
        IServiceProvider services,
        ILogger<RecurringJobsHostedService> logger)
    {
        _services = services;
        _logger = logger;
        
        _jobs = new List<RecurringJobDefinition>
        {
            new("session-cleanup", "0 * * * *", () => new CleanupExpiredSessions()),
            new("token-refresh", "*/15 * * * *", () => new RefreshExpiringTokens()),
            new("analytics-daily", "0 2 * * *", () => new AggregateDailyAnalytics(DateOnly.FromDateTime(DateTime.UtcNow.AddDays(-1)))),
        };
    }

    protected override async Task ExecuteAsync(CancellationToken stoppingToken)
    {
        // Initialize next run times
        foreach (var job in _jobs)
        {
            job.NextRun = CronExpression.Parse(job.CronExpression).GetNextOccurrence(DateTimeOffset.UtcNow);
        }

        while (!stoppingToken.IsCancellationRequested)
        {
            var now = DateTimeOffset.UtcNow;

            foreach (var job in _jobs.Where(j => j.NextRun <= now))
            {
                try
                {
                    _logger.LogInformation("Executing recurring job: {JobName}", job.Name);
                    
                    using var scope = _services.CreateScope();
                    var bus = scope.ServiceProvider.GetRequiredService<IMessageBus>();
                    
                    await bus.PublishAsync(job.MessageFactory());
                    
                    // Calculate next run
                    job.NextRun = CronExpression.Parse(job.CronExpression)
                        .GetNextOccurrence(DateTimeOffset.UtcNow);
                    
                    _logger.LogInformation("Job {JobName} next run: {NextRun}", job.Name, job.NextRun);
                }
                catch (Exception ex)
                {
                    _logger.LogError(ex, "Failed to execute recurring job: {JobName}", job.Name);
                }
            }

            // Sleep until next job is due (or max 1 minute)
            var nextRun = _jobs.Min(j => j.NextRun) ?? DateTimeOffset.UtcNow.AddMinutes(1);
            var delay = nextRun - DateTimeOffset.UtcNow;
            
            if (delay > TimeSpan.Zero)
            {
                await Task.Delay(
                    delay > TimeSpan.FromMinutes(1) ? TimeSpan.FromMinutes(1) : delay,
                    stoppingToken);
            }
        }
    }
}

public class RecurringJobDefinition
{
    public string Name { get; }
    public string CronExpression { get; }
    public Func<object> MessageFactory { get; }
    public DateTimeOffset? NextRun { get; set; }

    public RecurringJobDefinition(string name, string cronExpression, Func<object> messageFactory)
    {
        Name = name;
        CronExpression = cronExpression;
        MessageFactory = messageFactory;
    }
}

Great question. Let's break down the realistic limits and what PublyApp actually needs.

Wolverine + PostgreSQL Performance Characteristics
Wolverine uses PostgreSQL advisory locks + polling for message transport (not LISTEN/NOTIFY). Here's what that means:

Factor	Typical Limit	Notes
Polling interval	100ms - 1s default	Configurable, affects latency
Throughput	~1,000-5,000 msgs/sec	Single node, depends on hardware
Latency	100ms - 500ms	From enqueue to handler start
Concurrent workers	10-50 per instance	Configurable thread pool
Message size	Practical limit ~1MB	JSON serialized
What Does PublyApp Actually Need?
Let's do the math for a successful social media scheduler:

Assumptions (ambitious growth):

10,000 active users
Each user schedules ~5 posts/day average
Peak hour = 20% of daily posts
Daily volume:

10,000 users × 5 posts = 50,000 posts/day
Peak hour:

50,000 × 0.20 = 10,000 posts/hour
= ~167 posts/minute
= ~2.8 posts/second
Add system jobs:

Session cleanup: 1/hour
Token refresh: 4/hour  
Analytics: 1/day
Failed retry scans: 6/hour
Total peak load: ~3-5 messages/second

PostgreSQL can handle 1,000+ messages/second comfortably. You're at <1% capacity even with ambitious growth.

When Would You Hit Limits?
Users	Posts/Day	Peak msgs/sec	PostgreSQL Status
1,000	5,000	~0.3	😴 Sleeping
10,000	50,000	~3	✅ Easy
100,000	500,000	~30	✅ Fine
500,000	2,500,000	~150	⚠️ Monitor closely
1,000,000+	5,000,000+	~300+	🔄 Consider Redis
Reality check: At 100,000 users paying you money, you can afford to add Redis. That's a good problem to have.

## Bottleneck Hierarchy

When you do hit limits, here's the order they typically appear:
```
1. Database connections exhausted
   └─→ Solution: Connection pooling (PgBouncer)

2. Polling creates too much DB load  
   └─→ Solution: Increase poll interval or add Redis transport

3. Handler processing too slow
   └─→ Solution: Optimize handlers, add more instances

4. PostgreSQL write throughput maxed
   └─→ Solution: Bigger DB, read replicas, or Redis transport
```

    private readonly IServiceProvider _services;

    protected override async Task ExecuteAsync(CancellationToken ct)
    {
        while (!ct.IsCancellationRequested)
        {
            using var scope = _services.CreateScope();

            // Grab next job (with row lock)
            var job = await db.BackgroundJobs
                .FromSqlRaw("""
                    UPDATE background_jobs 
                    SET status = 'Processing', started_at = NOW()
                    WHERE id = (
                        SELECT id FROM background_jobs 
                        WHERE status = 'Pending' 
                        AND (scheduled_for IS NULL OR scheduled_for <= NOW())
                        ORDER BY created_at
                        FOR UPDATE SKIP LOCKED
                        LIMIT 1
                    )
                    RETURNING *
                    """)
                .FirstOrDefaultAsync(ct);

            if (job is not null)
            {
                await ProcessJobAsync(job, scope.ServiceProvider, ct);
            }
            else
            {
                await Task.Delay(1000, ct); // Poll interval
            }
        }
    }

    private async Task ProcessJobAsync(BackgroundJob job, IServiceProvider services, CancellationToken ct)
    {
        try
        {
            var handler = ResolveHandler(job.Type, services);
            await handler.HandleAsync(job.Payload, ct);

            job.Status = JobStatus.Completed;
            job.CompletedAt = DateTime.UtcNow;
        }
        catch (Exception ex)
        {
            job.RetryCount++;
            job.Error = ex.Message;
            job.Status = job.RetryCount >= 3 ? JobStatus.Failed : JobStatus.Pending;
            job.ScheduledFor = DateTime.UtcNow.AddSeconds(Math.Pow(2, job.RetryCount) * 10);
        }

        var db = services.GetRequiredService<AppDbContext>();
        await db.SaveChangesAsync(ct);
    }
}

1. The Outbox Pattern is NOT Optional
ChatGPT said "outbox_messages ... optional but recommended." This is critical, not optional.

Without it, you have a distributed transaction problem:

csharp
// DANGEROUS: What if this crashes between line 1 and 2?
await _db.SaveChangesAsync();           // 1. DB committed
await _bus.Publish(new PostClaimed());  // 2. Message sent (maybe)
If the process dies after line 1, the post is claimed but no message was sent. It's stuck forever.

Solution: Transactional outbox — message is written to DB in the same transaction, then a separate process publishes it.

csharp
// SAFE: Both succeed or both fail
await using var tx = await _db.Database.BeginTransactionAsync();

post.Status = PostStatus.Processing;
_db.OutboxMessages.Add(new OutboxMessage { Type = "PublishPost", Payload = ... });

await _db.SaveChangesAsync();
await tx.CommitAsync();
// A background process reads OutboxMessages and publishes to RabbitMQ

9. Graceful Shutdown
When you deploy, workers should finish their current job before stopping:

csharp
public class Worker : BackgroundService
{
    protected override async Task ExecuteAsync(CancellationToken stoppingToken)
    {
        while (!stoppingToken.IsCancellationRequested)
        {
            var job = await ClaimJobAsync();
            if (job == null)
            {
                await Task.Delay(1000, stoppingToken);
                continue;
            }
            
            // Don't pass stoppingToken here - let job finish
            await ProcessJobAsync(job, CancellationToken.None);
        }
    }
}

1. Stale Claim Recovery
With direct execution, if a worker crashes mid-job, the post is stuck in processing forever. You need a recovery mechanism:

csharp
public class RecoverStaleJobsJob : IJob
{
    public async Task Execute(IJobExecutionContext context)
    {
        // Jobs stuck in "processing" for more than 10 minutes = assumed crashed
        var staleThreshold = DateTime.UtcNow.AddMinutes(-10);
        
        var recovered = await _db.ScheduledPosts
            .Where(p => p.Status == PostStatus.Processing)
            .Where(p => p.ClaimedAt < staleThreshold)
            .ExecuteUpdateAsync(s => s
                .SetProperty(p => p.Status, PostStatus.Scheduled)
                .SetProperty(p => p.ClaimedAt, (DateTime?)null)
                .SetProperty(p => p.RetryCount, p => p.RetryCount + 1));

        if (recovered > 0)
            _logger.LogWarning("Recovered {Count} stale jobs", recovered);
    }
}
Schedule this every 5 minutes. Without it, crashed workers leave orphaned jobs.

2. Worker Pool Pattern (Even Without RabbitMQ)
If your Quartz job executes posts synchronously, you process one at a time. For parallelism without a broker:

csharp
public class PublishDuePostsJob : IJob
{
    private readonly SemaphoreSlim _concurrencyLimit = new(10); // Max 10 parallel

    public async Task Execute(IJobExecutionContext context)
    {
        var duePosts = await ClaimDuePostsAsync(batchSize: 50);
        
        var tasks = duePosts.Select(async post =>
        {
            await _concurrencyLimit.WaitAsync();
            try
            {
                await PublishPostAsync(post);
            }
            finally
            {
                _concurrencyLimit.Release();
            }
        });

        await Task.WhenAll(tasks);
    }
}
This gives you parallelism within a single worker process. Add this when you need throughput.

### 4. The "No Outbox" Trade-off

GPT said "jobs table is your outbox" — true, but there's a subtle difference:

**With MassTransit outbox:**
```
Transaction {
    Update business data
    Write to outbox
} → Commit
// Later: outbox publisher sends message
// Consumer executes independently
```

**With direct execution:**
```
Transaction {
    Claim job (update status)
} → Commit
// Then: execute API call
// Then: update status again
```

The difference: with direct execution, **the API call is outside the transaction**. If the API succeeds but the final status update fails, you have the idempotency problem we discussed.

This is why idempotency (Section 5 of my doc) is even MORE critical without a broker. You don't have message-level deduplication to save you.
